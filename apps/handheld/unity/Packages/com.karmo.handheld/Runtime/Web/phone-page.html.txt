(() => {
  'use strict';

  const $ = id => document.getElementById(id);
  const vf = $('vf'), vfCtx = vf.getContext('2d', { alpha: false, desynchronized: true });
  const chipMode = $('chipMode'), chipNet = $('chipNet'), chipPos = $('chipPos'), chipFrame = $('chipFrame'), chipOrient = $('chipOrient');
  const chipLens = $('chipLens'), chipWay = $('chipWay'), chipTrack = $('chipTrack');

  // 유니티는 공백·속도로 재정위를 **추론**하지만, WebXR 은 정답을 통보한다:
  // 기준 공간의 `reset` = 원점이 다시 잡혔다. 화면이 꺼진 동안엔 그 이벤트를 받을
  // rAF 자체가 안 도니 둘 다 필요하다. 카메라맨이 봐야 렌즈를 특징 많은 쪽으로 돌린다.
  let track = 'ok';            // ok | weak | lost | hidden | 3dof
  let wasEmulated = false, frozenPos = null;
  let trackResets = 0, trackSent = '', trackSentAt = 0;

  const TRACK_LABEL = {
    ok:     ['● 추적 양호', 0],
    weak:   ['▲ 추적 불안 — 위치가 추정값', 1],
    lost:   ['■ 추적 끊김 — 밝은 쪽·무늬 있는 곳으로', 1],
    hidden: ['■ 화면이 꺼졌다 — 켜 두세요', 1],
    '3dof': ['● 3DoF (회전만)', 0],
  };

  function setTrack(v) { if (v !== track) { track = v; paintTrack(); sendTrack(); } }

  // 우리 공백(=재정위)의 최대 원인이 **화면이 꺼져 rAF 가 멎는 것**이다. 실측 공백이
  // 439ms~15.6초였고 그 뒤 첫 프레임마다 좌표가 튀었다. 여기서 근원을 막는다.
  // 잠금은 화면을 안 보면 저절로 풀리므로 돌아올 때마다 다시 잡는다.
  let wakeLock = null;

  async function keepAwake() {
    if (!('wakeLock' in navigator)) { wakeAwake = false; return; }
    try {
      wakeLock = await navigator.wakeLock.request('screen');
      wakeAwake = true;
      wakeLock.addEventListener('release', () => { wakeAwake = false; wakeLock = null; });
    } catch (e) {
      wakeAwake = false;                       // 배터리 절약 모드 등에서 거절된다
      step.skip = 'wakelock ' + (e && e.message ? e.message : e);
    }
    paintTrack();
  }

  let wakeAwake = false;

  function dropAwake() {
    const w = wakeLock; wakeLock = null; wakeAwake = false;
    if (w) w.release().catch(() => {});
  }

  // 탭이 돌아오면 잠금이 풀려 있다 — 다시 잡는다.
  document.addEventListener('visibilitychange', () => {
    if (document.visibilityState === 'visible' && running && !wakeLock) keepAwake();
  });

  function paintTrack() {
    const [label, bad] = TRACK_LABEL[track] || [track, 1];
    chipTrack.className = 'chip' + (bad ? ' bad' : '');
    const awake = wakeAwake ? '' : ' · ⚠ 화면 꺼짐 방지 안 됨';
    chipTrack.innerHTML =
      (trackResets ? `<b>${label}</b> · 재정위 ${trackResets}회` : `<b>${label}</b>`) + awake;
    minTrack.className = bad ? 'bad' : 'ok';
    minTrack.textContent = label.slice(0, 1) + (trackResets ? ` ${trackResets}` : '');
  }

  // 유니티로: t|상태|재정위횟수|이번에재정위했나
  function sendTrack(reset) {
    if (!ws || ws.readyState !== 1) return;
    const line = ['t', track, trackResets, reset ? 1 : 0].join('|');
    if (!reset && line === trackSent && performance.now() - trackSentAt < 3000) return;
    trackSent = line; trackSentAt = performance.now();
    ws.send(line);
  }

  // ── 상태 ───────────────────────────────────────────────────────────────────
  let ws = null, wsReady = false;
  let mode = null;              // '6dof' | '3dof'
  let running = false;
  let seq = 0, rxFrames = 0, rxBytes = 0, txPoses = 0, statAt = performance.now();
  let pendingBitmap = false;

  // ── 쿼터니언 최소 도구 (three.js 없이) ───────────────────────────────────────
  const qMul = (a, b) => [
    a[3]*b[0] + a[0]*b[3] + a[1]*b[2] - a[2]*b[1],
    a[3]*b[1] - a[0]*b[2] + a[1]*b[3] + a[2]*b[0],
    a[3]*b[2] + a[0]*b[1] - a[1]*b[0] + a[2]*b[3],
    a[3]*b[3] - a[0]*b[0] - a[1]*b[1] - a[2]*b[2],
  ];
  const qAxis = (x, y, z, angle) => {
    const h = angle / 2, s = Math.sin(h);
    return [x*s, y*s, z*s, Math.cos(h)];
  };
  // three.js Euler 'YXZ' → quaternion
  const qEulerYXZ = (x, y, z) => {
    const c1 = Math.cos(x/2), c2 = Math.cos(y/2), c3 = Math.cos(z/2);
    const s1 = Math.sin(x/2), s2 = Math.sin(y/2), s3 = Math.sin(z/2);
    return [
      s1*c2*c3 + c1*s2*s3,
      c1*s2*c3 - s1*c2*s3,
      c1*c2*s3 - s1*s2*c3,
      c1*c2*c3 + s1*s2*s3,
    ];
  };

  // ── 소켓 ───────────────────────────────────────────────────────────────────
  function connect() {
    const proto = location.protocol === 'https:' ? 'wss:' : 'ws:';
    ws = new WebSocket(`${proto}//${location.host}/ws`);
    ws.binaryType = 'arraybuffer';
    ws.onopen = () => { wsReady = true; paint(); if (running) startRtc(); };
    ws.onclose = () => { wsReady = false; stopRtc(); paint(); if (running) setTimeout(connect, 800); };
    ws.onerror = () => { wsReady = false; paint(); };
    ws.onmessage = ev => {
      if (typeof ev.data === 'string') {
        // 왕복 재기 — 받은 눈금을 **그대로** 되돌린다. 폰이 아무 계산도 안 하므로
        // 두 시계를 맞출 필요가 없다(유니티가 같은 시계로 두 번 잰다).
        if (ev.data.charCodeAt(0) === 107 && ev.data[1] === '|') { ws.send(ev.data); return; }
        // WebRTC 시그널링
        if (ev.data.charCodeAt(0) === 119 && ev.data[1] === '|') { onRtcSignal(ev.data); return; }

        // 유니티가 되보내는 상태:
        //   s|x|y|z|yaw|fps|kb|틱Hz|캡처Hz|배율|초점거리mm|초점m|초점잡힘
        // 뒤 칸은 나중에 붙은 것이라 **길이를 보고** 읽는다 (옛 유니티에 붙어도 안 깨진다).
        const f = ev.data.split('|');
        if (f[0] === 's' && f.length >= 5)
          chipPos.innerHTML = `<b>${f[1]}</b> ${f[2]} <b>${f[3]}</b> · ${f[4]}°`;
        if (f[0] === 's' && f.length >= 13) {
          const shown = parseFloat(f[9]);
          // 유니티가 자른 값이 우리 목표와 크게 다르면(한계에 닿았다) 목표를 되맞춘다 —
          // 안 그러면 로커를 계속 밀어도 화면은 안 변하는데 숫자만 올라간다.
          if (isFinite(shown) && Math.abs(Math.log(shown / zoomTarget)) > 0.35) zoomTarget = shown;
          const hit = f[12] === '1';
          chipLens.className = 'chip' + (hit ? '' : ' bad');
          chipLens.innerHTML =
            `<b>${shown.toFixed(1)}×</b> ${Math.round(parseFloat(f[10]))}mm · 초점 <b>${parseFloat(f[11]).toFixed(1)}</b>m`;
          minLens.innerHTML = `<b>${shown.toFixed(1)}×</b> ${Math.round(parseFloat(f[10]))}mm`;
        }
        return;
      }
      rxFrames++; rxBytes += ev.data.byteLength;
      // WebRTC 영상이 붙었으면 JPEG 은 그리지 않는다 — 디코드 비용이 그대로 폰 CPU 다.
      // (유니티가 아직 보내고 있어도 여기서 끊는다. 끊김 대비로 경로는 살아 있다.)
      if (rtcVideo) return;
      if (pendingBitmap) return;             // 밀리면 그 장은 버린다 = 지연 누적 X
      pendingBitmap = true;
      createImageBitmap(new Blob([ev.data], { type: 'image/jpeg' }))
        .then(bmp => { drawFrame(bmp); bmp.close(); })
        .catch(() => {})
        .finally(() => { pendingBitmap = false; });
    };
  }

  function drawFrame(bmp) {
    const dpr = Math.min(devicePixelRatio || 1, 2);
    const w = Math.round(innerWidth * dpr), h = Math.round(innerHeight * dpr);
    if (vf.width !== w || vf.height !== h) { vf.width = w; vf.height = h; }

    // **레터박스(contain)** — 꽉 채우면(cover) 프레임 가장자리가 잘려 나가
    // 「보이는 것 ≠ 나가는 것」이 된다. 뷰파인더는 송출 프레임과 같아야 한다.
    vfCtx.fillStyle = '#000';
    vfCtx.fillRect(0, 0, w, h);

    // 화면만 돌리면 폰을 오른쪽으로 밀 때 그림이 위로 가는 **거짓 뷰파인더**가 된다.
    // 그래서 자동·가로 그립에서는 절대 안 돌린다. 세로 그립은 다르다 — 카메라 롤을
    // 90° 같이 돌려 보내므로, 화면도 같이 돌려야 겨냥 방향과 그림이 맞는다.
    const rotate = gripChoice === 'portrait' && (bmp.width >= bmp.height) !== (w >= h);
    const boxW = rotate ? h : w, boxH = rotate ? w : h;
    const s = Math.min(boxW / bmp.width, boxH / bmp.height);
    const dw = bmp.width * s, dh = bmp.height * s;

    vfCtx.save();
    if (rotate) {
      vfCtx.translate(w / 2, h / 2);
      vfCtx.rotate(-Math.PI / 2);
      vfCtx.translate(-boxW / 2, -boxH / 2);
    }
    vfCtx.drawImage(bmp, (boxW - dw) / 2, (boxH - dh) / 2, dw, dh);
    vfCtx.restore();

    // 탭한 화면 좌표를 **프레임 안 좌표**로 되돌리려면 방금 쓴 변환이 필요하다.
    // 레터박스 여백과 세로그립 90° 회전을 안 빼면 엉뚱한 데 초점이 잡힌다.
    lastDraw = { rotate, w, h, boxW, boxH, dw, dh,
                 ox: (boxW - dw) / 2, oy: (boxH - dh) / 2 };

    // 안내는 **자동 그립일 때만** 뜻이 있다 — 카메라맨이 세로를 골랐으면 그건 결정이지
    // 실수가 아니다. 자동일 때도 카메라 워킹 중 롤이 세로를 스쳐 가는 건 의도된 동작이라
    // **머무를 때만** 띄운다: 어긋남이 이어지면 켜고, 풀리면 즉시 끈다.
    // 「누워서」를 골랐는데 세워 들고 있을 때만 뜻이 있다. 「세워서」는 화면을 돌려
    // 채우므로 안내할 것이 없다.
    const mismatched = gripChoice === 'landscape' && (bmp.width >= bmp.height) !== (w >= h);
    if (!mismatched) { turnSince = 0; $('turnHint').classList.add('hidden'); }
    else {
      const now = performance.now();
      if (!turnSince) turnSince = now;
      if (now - turnSince > TURN_HINT_DELAY) $('turnHint').classList.remove('hidden');
    }

    if (frameW !== bmp.width || frameH !== bmp.height) {
      frameW = bmp.width; frameH = bmp.height;
      chipFrame.innerHTML = `<b>${bmp.width}×${bmp.height}</b>`;
    }
  }
  // 그립 = 카메라맨이 폰을 어떻게 잡을지 (송출 비율은 조종석이 정한다 — 별개다).
  // **시작할 때 한 번만** 고른다. 찍는 도중 바뀌면 송출 프레임이 90° 튀어 방송 사고가 된다.
  //   landscape : 눕혀 잡는다 = 롤 0, 16:9 가 그대로 꽉 찬다
  //   portrait  : 세워 잡는다 = 카메라 롤도 90° 돌려 수평을 맞추고 화면도 같이 돌린다
  let gripChoice = 'landscape';
  const gripRoll = () => (gripChoice === 'portrait' ? 90 : 0);

  let frameW = 0, frameH = 0, turnSince = 0;
  const TURN_HINT_DELAY = 2000;   // 이만큼 세로로 머물러야 안내가 뜬다

  // ── 렌즈 (줌 · 초점) ───────────────────────────────────────────────────────
  let lastDraw = null;
  let zoomTarget = 1, zoomRate = 0, zoomSentAt = 0, zoomLast = 1;
  const ZOOM_MIN = 0.5, ZOOM_MAX = 16;
  const ZOOM_MAX_RATE = 1.6;          // 로커 끝까지 밀었을 때 초당 e^1.6 ≈ 5배

  function sendZoom(force) {
    if (!ws || ws.readyState !== 1) return;
    const now = performance.now();
    if (!force && now - zoomSentAt < 33) return;         // 30Hz
    if (!force && Math.abs(zoomTarget - zoomLast) < 0.002) return;
    zoomSentAt = now; zoomLast = zoomTarget;
    ws.send(`z|${zoomTarget.toFixed(4)}`);
  }

  // 로커를 미는 동안 **로그 공간에서** 배율을 굴린다 — 1→2 와 8→16 이 같은 시간이 걸린다.
  // 선형으로 굴리면 망원에서 가속하는 것처럼 보인다.
  let zoomLoopAt = 0;
  function zoomLoop(t) {
    if (!running) { zoomLoopAt = 0; return; }
    requestAnimationFrame(zoomLoop);
    if (!zoomRate) { zoomLoopAt = t; return; }
    const dt = zoomLoopAt ? Math.min((t - zoomLoopAt) / 1000, 0.1) : 0;
    zoomLoopAt = t;
    zoomTarget = Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, zoomTarget * Math.exp(zoomRate * dt)));
    sendZoom(false);
  }

  function bindRocker() {
    const el = $('rocker'), fill = $('rockerFill');
    let id = null;

    const apply = e => {
      const r = el.getBoundingClientRect();
      // 가운데가 0, 위가 +(당김), 아래가 −(넓힘). 가운데 12% 는 데드존.
      let u = ((r.top + r.height / 2) - e.clientY) / (r.height / 2);
      u = Math.max(-1, Math.min(1, u));
      const dead = 0.12;
      const v = Math.abs(u) < dead ? 0 : Math.sign(u) * (Math.abs(u) - dead) / (1 - dead);
      zoomRate = v * ZOOM_MAX_RATE;
      fill.style.opacity = v ? '1' : '0';
      const half = r.height / 2 - 3;
      fill.style.height = Math.abs(v) * half + 'px';
      fill.style.top = v > 0 ? (half - Math.abs(v) * half + 3) + 'px' : (half + 3) + 'px';
    };

    el.addEventListener('pointerdown', e => {
      if (id !== null) return;
      id = e.pointerId; el.setPointerCapture(id); apply(e); e.preventDefault();
    });
    el.addEventListener('pointermove', e => { if (e.pointerId === id) { apply(e); e.preventDefault(); } });
    const off = e => {
      if (e.pointerId !== id) return;
      id = null; zoomRate = 0; fill.style.opacity = '0'; sendZoom(true);
    };
    el.addEventListener('pointerup', off);
    el.addEventListener('pointercancel', off);
  }
  bindRocker();

  $('btnZoom1').onclick = () => { zoomTarget = 1; sendZoom(true); };

  // 화면 좌표 → 프레임 안 뷰포트 좌표(0..1, **아래가 0** = 유니티 규약).
  // drawFrame 이 쓴 변환을 그대로 되돌린다. 프레임 밖(레터박스)을 찍으면 null.
  function screenToViewport(clientX, clientY) {
    if (!lastDraw) return null;
    const d = lastDraw;
    const dpr = d.w / innerWidth;
    let px = clientX * dpr, py = clientY * dpr;

    let bx, by;
    if (d.rotate) {
      // 그릴 때: translate(w/2,h/2) → rotate(-90°) → translate(-boxW/2,-boxH/2)
      const qx = px - d.w / 2, qy = py - d.h / 2;
      bx = -qy + d.boxW / 2;
      by = qx + d.boxH / 2;
    } else { bx = px; by = py; }

    const u = (bx - d.ox) / d.dw;
    const vTop = (by - d.oy) / d.dh;
    if (u < 0 || u > 1 || vTop < 0 || vTop > 1) return null;   // 검은 띠를 찍었다
    return [u, 1 - vTop];
  }

  function showFocusRing(clientX, clientY, ok) {
    const ring = $('focusRing');
    ring.style.left = clientX + 'px';
    ring.style.top = clientY + 'px';
    ring.classList.toggle('miss', !ok);
    ring.classList.remove('on');
    void ring.offsetWidth;                 // 애니메이션 다시 태우기
    ring.classList.add('on');
    clearTimeout(showFocusRing._t);
    showFocusRing._t = setTimeout(() => ring.classList.remove('on'), 900);
  }

  // 렌즈 판 — 한 손가락 탭 = 초점, 두 손가락 = 핀치 줌.
  function bindLens() {
    const el = $('lens');
    const pts = new Map();
    let pinchBase = 0, pinchZoom0 = 1, moved = 0, downAt = 0;

    el.addEventListener('pointerdown', e => {
      pts.set(e.pointerId, [e.clientX, e.clientY]);
      if (pts.size === 1) { moved = 0; downAt = performance.now(); }
      if (pts.size === 2) {
        const [a, b] = [...pts.values()];
        pinchBase = Math.hypot(a[0] - b[0], a[1] - b[1]) || 1;
        pinchZoom0 = zoomTarget;
      }
      el.setPointerCapture(e.pointerId);
      e.preventDefault();
    });

    el.addEventListener('pointermove', e => {
      const prev = pts.get(e.pointerId);
      if (!prev) return;
      if (pts.size === 1) moved += Math.hypot(e.clientX - prev[0], e.clientY - prev[1]);
      pts.set(e.pointerId, [e.clientX, e.clientY]);

      if (pts.size === 2) {
        const [a, b] = [...pts.values()];
        const d = Math.hypot(a[0] - b[0], a[1] - b[1]);
        zoomTarget = Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, pinchZoom0 * (d / pinchBase)));
        sendZoom(false);
      }
      e.preventDefault();
    });

    const up = e => {
      const wasSingle = pts.size === 1;
      pts.delete(e.pointerId);
      if (pts.size < 2) pinchBase = 0;
      if (pts.size === 0) sendZoom(true);

      // 탭 = 한 손가락 · 거의 안 움직임 · 짧게. 카메라 워킹 중 손이 스치는 걸 초점으로
      // 오해하면 방송 중에 초점이 튄다.
      if (!wasSingle || moved > 14 || performance.now() - downAt > 450) return;
      const vp = screenToViewport(e.clientX, e.clientY);
      if (!vp) { showFocusRing(e.clientX, e.clientY, false); return; }
      if (ws && ws.readyState === 1) ws.send(`f|${vp[0].toFixed(4)}|${vp[1].toFixed(4)}`);
      showFocusRing(e.clientX, e.clientY, true);
    };
    el.addEventListener('pointerup', up);
    el.addEventListener('pointercancel', e => { pts.delete(e.pointerId); });
  }
  bindLens();

  // ── WebRTC ─────────────────────────────────────────────────────────────────
  // 영상 = H.264 트랙(하드웨어 디코드) · 포즈 = DataChannel.
  // **MJPEG 은 안 지운다.** 안 붙거나 끊기면 그리로 돌아간다 — 방송 중에 길이 하나뿐이면
  // 그 길이 막힐 때 방송이 멈춘다.
  let pc = null, poseCh = null, rtcVideo = false, rtcWatch = 0;

  // WebRTC <video> 회전 — 캔버스와 같은 규칙(세로 그립 + 가로 그림 + 세로 화면).
  function applyVideoRotation() {
    const v = $('rtcv');
    const land = v.videoWidth >= v.videoHeight;
    const screenLand = innerWidth >= innerHeight;
    v.classList.toggle('rot', gripChoice === 'portrait' && land !== screenLand);
  }
  addEventListener('resize', () => { if (rtcVideo) applyVideoRotation(); });

  // 영상만 내린다 (데이터채널은 그대로) — MJPEG 이 다시 그리기 시작한다.
  function hideRtcVideo() {
    const v = $('rtcv');
    v.classList.remove('on');
    rtcVideo = false;
    paintWay();
  }

  const usingRtc = () => poseCh && poseCh.readyState === 'open';

  function paintWay() {
    const v = rtcVideo ? 'WebRTC' : (usingRtc() ? 'WebRTC(포즈만)' : 'MJPEG');
    chipWay.className = 'chip' + (rtcVideo ? '' : ' bad');
    chipWay.innerHTML = `<b>${v}</b>`;
  }

  async function startRtc() {
    if (pc || !ws || ws.readyState !== 1) return;
    try {
      pc = new RTCPeerConnection({
        iceServers: [{ urls: 'stun:stun.l.google.com:19302' }],
      });

      // 포즈는 **늦으면 버린다** — 재전송을 기다리면 그건 정보가 아니라 지연이다.
      // (유니티 쪽 큐 넷이 이미 같은 정신으로 서 있다.)
      poseCh = pc.createDataChannel('pose', { ordered: false, maxRetransmits: 0 });
      poseCh.onopen = () => { paintWay(); };
      poseCh.onclose = () => { paintWay(); };

      pc.ontrack = (e) => {
        const v = $('rtcv');
        if (e.streams && e.streams[0] && v.srcObject !== e.streams[0]) v.srcObject = e.streams[0];

        // ★ **트랙이 왔다 != 그림이 왔다** (2026-08-21 실측으로 고침).
        //   우리가 recvonly 자리를 만들어 두면, 유니티가 영상 트랙을 안 붙였을 때도
        //   브라우저는 ontrack 을 띄운다 — 프레임이 영영 안 오는 빈 트랙이다.
        //   그걸 믿고 <video> 를 띄우면 검은 판이 뷰파인더를 덮고, rtcVideo 를 켠 탓에
        //   MJPEG 도 안 그려서 화면이 통째로 검어진다. 실제로 그렇게 됐다.
        //   그래서 **videoWidth 가 잡힐 때만** 띄운다.
        const show = () => {
          if (v.videoWidth > 0 && !rtcVideo) {
            v.classList.add('on');
            // ★ 캔버스(MJPEG)와 **같은 규칙**으로 돌린다. 규칙이 하나면 두 경로가 안 갈린다.
            //   이 클래스는 정의만 돼 있고 붙이는 곳이 없었다 — 그래서 WebRTC 로 보면
            //   세로 그립인데 그림이 안 돌아 옆으로 누웠다.
            applyVideoRotation();
            rtcVideo = true; paintWay();
          }
        };
        v.onloadedmetadata = show;
        v.onresize = show;
        v.onplaying = show;
        show();

        // 그림이 안 오면 되돌린다 — 「붙었다」고 믿고 기다리는 검은 화면보다 MJPEG 이 낫다.
        clearTimeout(rtcWatch);
        rtcWatch = setTimeout(() => { if (v.videoWidth === 0) hideRtcVideo(); }, 4000);

        const t = e.track;
        if (t) {
          t.onended = hideRtcVideo;
          t.onmute = hideRtcVideo;
        }
      };
      pc.onicecandidate = (e) => {
        if (e.candidate && ws && ws.readyState === 1) {
          const c = e.candidate;
          ws.send(`w|ice|${c.candidate}|${c.sdpMid}|${c.sdpMLineIndex}`);
        }
      };
      pc.onconnectionstatechange = () => {
        if (!pc) return;
        if (pc.connectionState === 'failed' || pc.connectionState === 'disconnected') stopRtc();
      };

      // 영상을 **받기만** 한다 (폰 카메라는 안 보낸다 — AR 이 쓰고 있다).
      pc.addTransceiver('video', { direction: 'recvonly' });

      const offer = await pc.createOffer();
      await pc.setLocalDescription(offer);
      ws.send('w|offer|' + offer.sdp);
    } catch (e) {
      stopRtc();
    }
  }

  function stopRtc() {
    clearTimeout(rtcWatch);
    rtcVideo = false;
    const v = $('rtcv');
    v.classList.remove('on');
    v.srcObject = null;
    try { poseCh && poseCh.close(); } catch (e) {}
    try { pc && pc.close(); } catch (e) {}
    poseCh = null; pc = null;
    paintWay();                       // MJPEG 으로 돌아간다
  }

  async function onRtcSignal(line) {
    if (!pc) return;
    if (line.startsWith('w|answer|')) {
      await pc.setRemoteDescription({ type: 'answer', sdp: line.slice('w|answer|'.length) });
      return;
    }
    if (line.startsWith('w|ice|')) {
      const t = line.split('|');
      if (t.length < 5) return;
      try {
        await pc.addIceCandidate({ candidate: t[2], sdpMid: t[3], sdpMLineIndex: +t[4] });
      } catch (e) {}
    }
  }

  // 포즈가 어디서 멎는지 세는 발자국. 「안 온다」만으론 XR 루프가 안 도는 건지,
  // 돌지만 자세를 못 얻는 건지, 얻었는데 못 보내는 건지 가를 수 없다.
  const step = { frame: 0, pose: 0, sent: 0, skip: '' };

  function sendPose(t, px, py, pz, qx, qy, qz, qw, fovDeg, aspect, dof) {
    step.pose++;
    const line = `p|${t.toFixed(1)}|${f5(px)}|${f5(py)}|${f5(pz)}|${f5(qx)}|${f5(qy)}|${f5(qz)}|${f5(qw)}|${fovDeg.toFixed(2)}|${aspect.toFixed(4)}|${dof}|${gripRoll()}`;

    // 데이터채널이 열려 있으면 그리로. 규약은 **똑같다** — 길이 둘이어도 문법은 하나다.
    if (usingRtc()) {
      // 밀려 있으면 건너뛴다(WS 의 bufferedAmount 게이트와 같은 규칙).
      if (poseCh.bufferedAmount > 32 * 1024) { step.skip = 'dc-buf'; return; }
      poseCh.send(line);
      step.sent++; txPoses++; seq++;
      return;
    }

    if (!ws) { step.skip = 'no-ws'; return; }
    if (ws.readyState !== 1) { step.skip = 'ws-state' + ws.readyState; return; }
    if (ws.bufferedAmount > 32 * 1024) { step.skip = 'ws-buf'; return; }
    ws.send(line);
    step.sent++; txPoses++; seq++;
  }

  const f5 = n => n.toFixed(5);

  // ── 6DoF: WebXR immersive-ar ───────────────────────────────────────────────
  let xrSession = null, xrWay = '';

  async function startXR() {
    const gl = $('gl').getContext('webgl', { xrCompatible: true, alpha: true, antialias: false });
    // ★ **덜 요구하는 쪽으로 차례로 물어본다** (2026-08-21).
    //   "The specified session configuration is not supported" 는 **어느 기능이** 막혔는지
    //   안 말한다. 한 벌만 던지고 실패하면 거기서 끝이라, 기기가 뭘 줄 수 있는지 영영 모른다.
    //   그래서 요구를 하나씩 덜어 가며 묻고, **어디서 붙었는지를 화면에 남긴다.**
    //   dom-overlay 가 없으면 우리 UI 가 AR 화면 위에 안 뜨지만 **포즈는 그대로 온다** —
    //   손카메라의 본체는 포즈이므로, 못 쓰는 것보다 낫다.
    const CONFIGS = [
      { name: 'local + dom-overlay', opts: {
          requiredFeatures: ['local'], optionalFeatures: ['dom-overlay'],
          domOverlay: { root: $('overlay') } } },
      { name: 'local (오버레이 없이)', opts: { requiredFeatures: ['local'] } },
      { name: 'viewer', opts: { requiredFeatures: ['viewer'] } },
      { name: '요구 없음', opts: {} },
    ];

    const tried = [];
    let used = null;
    for (const c of CONFIGS) {
      try {
        xrSession = await navigator.xr.requestSession('immersive-ar', c.opts);
        used = c.name;
        break;
      } catch (e) {
        tried.push(`${c.name}: ${e && e.name ? e.name : ''} ${e && e.message ? e.message : e}`);
      }
    }
    if (!xrSession) {
      // 전부 실패 — 무엇을 물었고 각각 뭐라 했는지를 그대로 남긴다. 추측하지 않는다.
      const err = new Error('AR 세션을 못 열었다:\n' + tried.join('\n'));
      err.tried = tried;
      throw err;
    }
    xrWay = used;

    await gl.makeXRCompatible();
    xrSession.updateRenderState({ baseLayer: new XRWebGLLayer(xrSession, gl) });

    // 기준 공간도 같은 방식 — local 이 안 되면 덜 요구하는 쪽으로.
    // (local 이 없으면 위치가 안 오고 회전만 온다 = 3DoF 와 같아진다. 그것도 화면에 적는다.)
    let refSpace = null, spaceName = null;
    for (const nm of ['local', 'local-floor', 'viewer']) {
      try { refSpace = await xrSession.requestReferenceSpace(nm); spaceName = nm; break; }
      catch (e) { tried.push(`refSpace ${nm}: ${e && e.message ? e.message : e}`); }
    }
    if (!refSpace) throw new Error('기준 공간을 못 얻었다:\n' + tried.join('\n'));
    xrWay += ' · ' + spaceName;

    // 원점이 다시 잡혔다는 **플랫폼의 정답 통보**. 유니티는 이걸 확정 재정위로 쓴다
    // (공백·속도 판정은 이 이벤트를 못 받는 경우를 위한 그물로 남는다).
    if (refSpace.addEventListener)
      refSpace.addEventListener('reset', () => { trackResets++; paintTrack(); sendTrack(true); });

    // 화면을 안 보고 있으면 rAF 가 멈춘다 — 우리 공백의 대부분이 이것이다.
    xrSession.addEventListener('visibilitychange', () =>
      setTrack(xrSession && xrSession.visibilityState === 'visible' ? 'ok' : 'hidden'));

    // 포즈는 rAF 마다 하나다 — 프레임률이 곧 포즈율이다. 기기가 더 줄 수 있으면 받는다.
    try {
      const rates = xrSession.supportedFrameRates;
      if (rates && rates.length && xrSession.updateTargetFrameRate) {
        const best = Math.max.apply(null, Array.from(rates));
        xrSession.updateTargetFrameRate(best).catch(() => {});
        xrWay += ' · ' + best + 'Hz';
      }
    } catch (e) { /* 기기가 안 받아 주면 그냥 기본값으로 돈다 */ }

    mode = '6dof';
    enterRunning();

    xrSession.addEventListener('end', () => { xrSession = null; stop(); });

    const onFrame = (t, frame) => {
      if (!xrSession) return;
      // ★ 다음 프레임 예약을 **맨 먼저** 한다. 아래에서 던지면 루프가 그대로 끊기는데,
      //   그러면 「포즈가 안 온다」로만 보이고 원인이 안 남는다.
      xrSession.requestAnimationFrame(onFrame);
      try {
      step.frame++;
      const pose = frame.getViewerPose(refSpace);
      if (!pose || !pose.views.length) { step.skip = 'no-viewerpose'; setTrack('lost'); return; }

      // emulatedPosition = 위치가 실측이 아니라 **추정**이다. 스펙이 말하길 이 구간에는
      // 세상이 따라 끌려오고(world drags along), 추적이 돌아오는 첫 프레임에 위치가
      // **홱 뛴다**. 그래서 ① 추정 구간에는 마지막 실측 위치를 얼려 보내고
      // ② 돌아오는 순간을 재정위로 통보한다(유니티가 그 계단을 흡수한다).
      const emulated = !!pose.emulatedPosition;
      if (emulated !== wasEmulated) {
        if (wasEmulated) { trackResets++; paintTrack(); sendTrack(true); }   // 추정 -> 실측
        wasEmulated = emulated;
      }
      setTrack(emulated ? 'weak' : 'ok');
      const view = pose.views[0];
      const o = view.transform.orientation;
      let p = view.transform.position;
      if (emulated) p = frozenPos || p;         // 추정 구간 = 얼린 자리
      else frozenPos = { x: p.x, y: p.y, z: p.z };

      // 투영행렬에서 실제 화각을 뽑는다 = 폰 카메라의 진짜 FOV
      // 비대칭 절두체일 수 있으므로 위·아래를 따로 재서 더한다
      const m = view.projectionMatrix;
      const fovY = (Math.atan((1 + m[9]) / m[5]) + Math.atan((1 - m[9]) / m[5])) * 180 / Math.PI;
      const vp = xrSession.renderState.baseLayer.getViewport(view);
      const aspect = vp && vp.height ? vp.width / vp.height : innerWidth / innerHeight;

      sendPose(t, p.x, p.y, p.z, o.x, o.y, o.z, o.w, fovY, aspect, '6');
      tick();
      } catch (e) { step.skip = 'frame-err ' + (e && e.message ? e.message : e); }
    };
    xrSession.requestAnimationFrame(onFrame);
  }

  // ── 3DoF 폴백: DeviceOrientation ───────────────────────────────────────────
  const qFlip = qAxis(1, 0, 0, -Math.PI / 2);   // 「폰을 눕힌 자세」를 정면으로
  let devQuat = [0, 0, 0, 1], gotOrientation = false;

  function onDeviceOrientation(e) {
    if (e.alpha === null) return;
    const d = Math.PI / 180;
    const orient = ((screen.orientation && screen.orientation.angle) || window.orientation || 0) * d;
    let q = qEulerYXZ(e.beta * d, e.alpha * d, -e.gamma * d);
    q = qMul(q, qFlip);
    q = qMul(q, qAxis(0, 0, 1, -orient));
    devQuat = q;
    gotOrientation = true;
  }

  function start3Dof() {
    mode = '3dof';
    setTrack('3dof');
    addEventListener('deviceorientation', onDeviceOrientation, true);
    enterRunning();
    const loop = t => {
      if (!running || mode !== '3dof') return;
      requestAnimationFrame(loop);
      if (!gotOrientation) return;
      const q = devQuat;
      sendPose(t, 0, 0, 0, q[0], q[1], q[2], q[3], 60, innerWidth / innerHeight, '3');
      tick();
    };
    requestAnimationFrame(loop);
  }

  // ── 조이스틱 ───────────────────────────────────────────────────────────────
  // 누른 자리가 곧 스틱의 중심이 된다 — 고정 패드보다 손이 어디에 있든 바로 잡힌다.
  const stick = { l: [0, 0], r: [0, 0] };
  let stickDirty = false, stickTimer = null;
  const RADIUS = 52;                       // 이 거리에서 1.0

  function bindStick(zoneId, ringId, key) {
    const zone = $(zoneId), ring = $(ringId), knob = ring.firstElementChild;
    let id = null, ox = 0, oy = 0;

    const set = (x, y) => {
      stick[key] = [x, y];
      stickDirty = true;
      knob.style.transform = `translate(${x * RADIUS}px, ${-y * RADIUS}px)`;
    };

    zone.addEventListener('pointerdown', e => {
      if (id !== null) return;
      id = e.pointerId;
      zone.setPointerCapture(id);
      const r = zone.getBoundingClientRect();
      ox = e.clientX; oy = e.clientY;
      ring.style.left = (ox - r.left) + 'px';
      ring.style.top = (oy - r.top) + 'px';
      ring.classList.add('on');
      set(0, 0);
      e.preventDefault();
    });

    zone.addEventListener('pointermove', e => {
      if (e.pointerId !== id) return;
      let dx = (e.clientX - ox) / RADIUS;
      let dy = (oy - e.clientY) / RADIUS;          // 화면 y 는 아래가 +
      const len = Math.hypot(dx, dy);
      if (len > 1) { dx /= len; dy /= len; }       // 원 밖으로 안 나간다
      set(dx, dy);
      e.preventDefault();
    });

    const release = e => {
      if (e.pointerId !== id) return;
      id = null;
      ring.classList.remove('on');
      set(0, 0);
    };
    zone.addEventListener('pointerup', release);
    zone.addEventListener('pointercancel', release);
  }

  function sendSticks() {
    if (!stickDirty) return;
    if (!ws || ws.readyState !== 1) return;
    stickDirty = false;
    const f = n => n.toFixed(3);
    ws.send(`j|${f(stick.l[0])}|${f(stick.l[1])}|${f(stick.r[0])}|${f(stick.r[1])}`);
  }

  bindStick('zoneL', 'stickL', 'l');
  bindStick('zoneR', 'stickR', 'r');

  // ── 공통 ───────────────────────────────────────────────────────────────────
  function enterRunning() {
    // 여기서 화면 방향이 잠긴다. 고른 그립과 실제 방향이 어긋나면 **현실을 따른다** —
    // 가로로 들어갔는데 롤을 90° 걸면 그림이 옆으로 눕는다.
    if (innerWidth >= innerHeight) gripChoice = 'landscape';
    running = true;
    $('start').classList.add('gone');
    keepAwake();
    $('hudMin').classList.remove('hidden');
    $('hud').classList.remove('hidden');
    $('hud').classList.add('folded');
    paintTrack();
    $('bar').classList.remove('hidden');
    $('sticks').classList.remove('hidden');
    $('lens').classList.remove('hidden');
    $('rocker').classList.remove('hidden');
    $('btnZoom1').classList.remove('hidden');
    if (stickTimer === null) stickTimer = setInterval(sendSticks, 33);   // 30 Hz
    if (!zoomLoopAt) requestAnimationFrame(zoomLoop);
    if (!ws) connect(); else startRtc();
    paint();
    paintWay();
    // 폰이 자기 상태를 서버에 알린다 — 유니티 콘솔은 밖에서 못 읽고 조종석은 사람 눈에만
    // 보이므로, /diag 로 **양쪽**을 한 자리에서 보려면 폰 쪽 사실도 올라가야 한다.
    sendDiag();
    if (diagTimer === null) diagTimer = setInterval(sendDiag, 5000);
  }

  let diagTimer = null;

  function sendDiag() {
    if (!ws || ws.readyState !== 1) return;
    const o = (screen.orientation && screen.orientation.type) || '?';
    const a = (screen.orientation && screen.orientation.angle);
    const way = rtcVideo ? 'webrtc' : (usingRtc() ? 'webrtc-pose' : 'mjpeg');
    // 칸에 | 가 들어가면 규약이 깨진다 — 안전한 글자만 보낸다.
    const safe = v => String(v).replace(/[|]/g, '/');
    sendTrack();
    ws.send(['d', safe(mode || '-'), safe(xrWay || '-'), safe(gripChoice),
             safe(o), safe(a), `${innerWidth}x${innerHeight}`, safe(way),
             safe(frameW + 'x' + frameH),
             `f${step.frame}/p${step.pose}/s${step.sent}`, safe(step.skip || '-')].join('|'));
  }

  function stop() {
    running = false;
    if (diagTimer !== null) { clearInterval(diagTimer); diagTimer = null; }
    removeEventListener('deviceorientation', onDeviceOrientation, true);
    if (xrSession) { const s = xrSession; xrSession = null; s.end().catch(() => {}); }
    $('start').classList.remove('gone');
    dropAwake();
    $('hudMin').classList.add('hidden');
    $('hud').classList.add('hidden');
    $('bar').classList.add('hidden');
    $('sticks').classList.add('hidden');
    $('lens').classList.add('hidden');
    $('rocker').classList.add('hidden');
    $('btnZoom1').classList.add('hidden');
    $('turnHint').classList.add('hidden');
    zoomRate = 0; zoomLoopAt = 0;
    turnSince = 0;
    if (stickTimer !== null) { clearInterval(stickTimer); stickTimer = null; }
    stick.l = [0, 0]; stick.r = [0, 0];
    if (ws && ws.readyState === 1) { ws.send('j|0|0|0|0'); ws.send('w|bye'); }   // 놓고 나가도 안 밀린다
    stopRtc();
  }

  function tick() {
    const now = performance.now();
    if (now - statAt < 500) return;
    const dt = (now - statAt) / 1000;
    const fps = rxFrames / dt, kbs = rxBytes / 1024 / dt, hz = txPoses / dt;
    rxFrames = rxBytes = txPoses = 0; statAt = now;
    chipNet.className = 'chip' + (wsReady ? '' : ' bad');
    chipNet.innerHTML = wsReady
      ? `뷰파인더 <b>${fps.toFixed(0)}</b> fps · ${kbs.toFixed(0)} KB/s · 포즈 <b>${hz.toFixed(0)}</b> Hz`
      : '<b>끊김</b> — 다시 붙는 중';
    minNet.innerHTML = wsReady ? `<b>${fps.toFixed(0)}</b> fps` : '<b>끊김</b>';
  }

  // 화면이 도는지 재서 보여준다 — AR 세션은 회전을 잠글 수 있어서, 기기 자동회전을
  // 켜도 페이지가 안 도는 경우가 있다. 추측 대신 이 값을 보고 판단한다.
  function paintOrient() {
    const a = (screen.orientation && screen.orientation.angle);
    const t = (screen.orientation && screen.orientation.type) || '?';
    chipOrient.innerHTML =
      `<b>${innerWidth}×${innerHeight}</b> ∠${a === undefined ? '?' : a} ${t.replace('-primary','').replace('-secondary','₂')}`;
  }
  addEventListener('resize', paintOrient);
  if (screen.orientation) screen.orientation.addEventListener('change', paintOrient);
  setInterval(paintOrient, 500);
  paintOrient();

  // AR 세션은 **진입 시점의 화면 방향으로 잠긴다** — 들어간 뒤 폰을 돌려도 페이지는
  // 안 돈다(2026-08-20 실기 확인: 가로로 들어도 ∠0 portrait). 그래서 「눕혀서」로
  // 쓰려면 눕힌 채로 시작을 눌러야 한다. 시작 화면에서 그걸 말해 준다.
  function paintOrientNote() {
    if (running) return;
    const note = $('orientNote');
    const landscape = innerWidth >= innerHeight;
    if (gripChoice === 'landscape' && !landscape) {
      note.className = 'warn';
      note.innerHTML = '⚠ 지금 <b>세로</b>다. 화면 방향은 <b>시작하는 순간으로 잠긴다</b> — ' +
                       '폰을 눕힌 뒤에 시작을 눌러라.';
    } else if (gripChoice === 'portrait' && landscape) {
      note.className = 'warn';
      note.innerHTML = '⚠ 지금 <b>가로</b>다. 세워서 잡을 거면 폰을 세운 뒤에 시작을 눌러라.';
    } else {
      note.className = 'ok';
      note.innerHTML = landscape
        ? '가로로 들어간다 — 16:9 가 꽉 찬다.'
        : '세로로 들어간다 — 화면을 돌려 16:9 를 채운다.';
    }
  }
  setInterval(paintOrientNote, 400);
  addEventListener('resize', paintOrientNote);

  function paint() {
    chipMode.innerHTML = mode === '6dof' ? `<b>6DoF</b> ${xrWay || 'WebXR'}`
      : mode === '3dof' ? '<b>3DoF</b> 자이로 (회전만)' : '대기';
    chipNet.className = 'chip' + (wsReady ? '' : ' bad');
  }

  // ── 진입 ───────────────────────────────────────────────────────────────────
  for (const [id, g] of [['gripLand', 'landscape'], ['gripPort', 'portrait']]) {
    $(id).onclick = () => {
      gripChoice = g;
      for (const other of ['gripLand', 'gripPort']) {
        const on = (other === id);
        $(other).classList.toggle('sel', on);
        $(other).setAttribute('aria-checked', String(on));
      }
      paintOrientNote();
    };
  }
  $('btnStop').onclick = stop;
  $('btnRecenter').onclick = () => { if (ws && ws.readyState === 1) ws.send('c|recenter'); };
  $('btnFallback').onclick = () => start3Dof();

  (async () => {
    const note = $('modeNote'), btn = $('btnStart');
    let xrOk = false;
    try { xrOk = !!(navigator.xr && await navigator.xr.isSessionSupported('immersive-ar')); }
    catch (e) { xrOk = false; }

    if (location.protocol !== 'https:' && location.hostname !== 'localhost') {
      note.textContent = 'https 가 아니라 센서를 못 쓴다 — cloudflared 터널 주소로 열어라.';
      btn.textContent = 'https 필요';
      return;
    }

    if (xrOk) {
      btn.disabled = false;
      btn.textContent = '시작 (6DoF)';
      note.textContent = 'ARCore 로 위치까지 따라간다. 시작을 누르면 권한을 묻는다.';
      $('btnFallback').style.display = '';
      btn.onclick = () => {
        btn.disabled = true;
        startXR().catch(err => {
          btn.disabled = false;
          // 여러 줄로 보여 준다 — 어느 설정이 왜 거부됐는지가 그 안에 있다.
          note.style.whiteSpace = 'pre-wrap';
          note.style.textAlign = 'left';
          note.style.maxWidth = '90vw';
          note.textContent = 'WebXR 실패\n' + (err && err.message || err)
            + '\n\nARCore(Google Play 서비스 (AR))가 최신인지, 다른 앱이 카메라를 잡고 있지 않은지 보라.';
          $('btnFallback').style.display = '';
        });
      };
    } else {
      btn.disabled = false;
      btn.textContent = '시작 (자이로 · 회전만)';
      note.textContent = 'WebXR(ARCore)이 없다 — 회전만 전달된다. Play 스토어의 「Google Play 서비스 (AR)」를 깔면 6DoF 가 된다.';
      btn.onclick = () => start3Dof();
    }
  })();
})();
